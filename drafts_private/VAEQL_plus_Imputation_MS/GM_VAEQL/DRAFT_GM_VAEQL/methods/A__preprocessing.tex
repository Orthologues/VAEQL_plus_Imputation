\begin{algorithm}[ht]
\footnotesize
\caption{Feature-Type-Specific Preprocessing, Synthetic Amputation, and Placeholder Filling}
\label{algo:data_preprocessing}
\begin{algorithmic}[1]% Using [1] for line numbering every line is usually clearer
\setlength{\itemsep}{0.3em}
\Require Tabular data $\mathcal D$ and feature schema $\mathcal F$;
         missing mechanism $\kappa \in\{\mathrm{MCAR},\mathrm{MAR},\mathrm{MNAR}\}$;
         amputation rate $0<\rho\leq 0.5$
\Ensure Model-input Tensor $X_0$ and Ordered Feature Specifications $\Phi$;
        Missingness Mask $M$ retained separately

\Statex
\Procedure{Preprocess}{$\mathcal D,\mathcal F,\kappa,\rho$}
  \State Validate that $\mathcal D$ contains every feature specified by $\mathcal F$
  \State $X_{\mathrm{real}}\gets
         \textsc{ZScoreAndClip}(\mathcal D,\mathcal F_{\mathrm{real}})$
         \Comment{standardize observed values; clip to $[-5,5]$}
  \State $X_{\mathrm{pos}}\gets
         \textsc{YeoJohnsonAndZScoreAndClip}(\max(\mathcal D_{\mathrm{pos}},0))$
         \Comment{transform and standardize observed values; clip to $[-5,5]$}
  \State $X_{\mathrm{count}}\gets
         \operatorname{clip}_{[-5,5]}\!\left(\log\!\left(1+\max(\mathcal D_{\mathrm{count}},0)\right)\right)$
  \State $(X_{\mathrm{ord}},G_{\mathrm{ord}})\gets
         \textsc{UnaryEncode}(\mathcal D,\mathcal F_{\mathrm{ord}})$
         \Comment{one column per ordinal threshold}
  \State $(X_{\mathrm{bin}},G_{\mathrm{bin}})\gets
         \textsc{BinaryEncode}(\mathcal D,\mathcal F_{\mathrm{bin}})$
  \State $(X_{\mathrm{cat}},G_{\mathrm{cat}})\gets
         \textsc{OneHotEncode}(\mathcal D,\mathcal F_{\mathrm{cat}})$
         \Comment{preserve missing source rows}

  \State $P\gets\textsc{ConcatenateByFeatureType}
         (X_{\mathrm{real}},X_{\mathrm{pos}},X_{\mathrm{count}},
          X_{\mathrm{ord}},X_{\mathrm{bin}},X_{\mathrm{cat}})$
  \State $\mathcal J_{\mathrm{num}}\gets
         \operatorname{columns}(X_{\mathrm{real}})
         \cup\operatorname{columns}(X_{\mathrm{pos}})
         \cup\operatorname{columns}(X_{\mathrm{count}})$
  \State $\mathcal G\gets
         \{G_{\mathrm{ord}}(f):f\in\mathcal F_{\mathrm{ord}}\}
         \cup\{\{b\}:b\in\operatorname{columns}(X_{\mathrm{bin}})\}
         \cup\{G_{\mathrm{cat}}(c):c\in\mathcal F_{\mathrm{cat}}\}
         \cup\{\{j\}:j\in\mathcal J_{\mathrm{num}}\}$
         \Comment{ordinal/categorical groups; binary/numeric singletons}
  \State $(A,M)\gets\textsc{ApplyPrefillAndPyAmpute}
         (P,\mathcal G,G_{\mathrm{ord}},G_{\mathrm{bin}},G_{\mathrm{cat}},\kappa,\rho)$
  \State $A_{\mathrm{filled}}\gets\textsc{FillAmputedPlaceholders}
         (A,M,G_{\mathrm{ord}},G_{\mathrm{bin}},G_{\mathrm{cat}})$

  \State $\Phi\gets\textsc{BuildOrderedFeatureSpecs}
         (\operatorname{columns}(A_{\mathrm{filled}});
          \operatorname{columns}(X_{\mathrm{real}}),
          \operatorname{columns}(X_{\mathrm{pos}}),
          \operatorname{columns}(X_{\mathrm{count}}),
          G_{\mathrm{ord}},\operatorname{columns}(X_{\mathrm{bin}}),G_{\mathrm{cat}})$
         \Comment{numeric/binary singletons; ordinal/categorical groups}
  \State $X_0\gets A_{\mathrm{filled}}$
  \State \Return $(X_0,\Phi,M)$
\EndProcedure

\Statex
\Procedure{ApplyPrefillAndPyAmpute}{$P,\mathcal G,G_{\mathrm{ord}},G_{\mathrm{bin}},G_{\mathrm{cat}},\kappa,\rho$}
  \State $M'\gets\operatorname{isnan}(P)$
         \Comment{record source missingness after feature transformation}
  \State $\mathbf v_{\mathrm{pre}}\gets
         \textsc{FeatureAwareMarginals}(P;M'=0,G_{\mathrm{ord}},G_{\mathrm{bin}},G_{\mathrm{cat}})$
         \Comment{using: numeric medians; binary prevalence; ordinal threshold rates; categorical proportions}
  \State $\widetilde P\gets\textsc{FillSourceMissing}(P,\mathbf v_{\mathrm{pre}})$
         \Comment{complete input required by Pyampute}
  \State $A\gets\widetilde P$
  \For{each group $g\in\mathcal G$}
    \State $Y\gets\textsc{Pyampute}
           (\widetilde P;\ \texttt{incomplete\_vars}=g,
            \texttt{mechanism}=\kappa,\ \texttt{prop}=\rho)$
    \State $A_{:,g}\gets Y_{:,g}$
  \EndFor
  \State $\widetilde M\gets\operatorname{isnan}(A)$
  \State $M\gets 0$; set $M[M']\gets 1$ and
         $M[\widetilde M\land\neg M']\gets 2$
         \Comment{$1$: source-missing; $2$: newly amputed}
  \State \Return $(A,M)$
\EndProcedure

\Statex
\Procedure{FillAmputedPlaceholders}{$A,M,G_{\mathrm{ord}},G_{\mathrm{bin}},G_{\mathrm{cat}}$}
  \State $\Omega_2\gets\{(i,j):M_{ij}=2\}$
         \Comment{fill newly amputed cells only; retain source-missing placeholders}
  \If{$\Omega_2=\varnothing$}
    \State \Return $A$
  \EndIf
  \State $\mathbf v_0\gets
         \textsc{FeatureAwareMarginals}(A;M=0,G_{\mathrm{ord}},G_{\mathrm{bin}},G_{\mathrm{cat}})$
         \Comment{using: numeric medians; binary prevalence; ordinal threshold rates; categorical proportions}
  \State $A_{\mathrm{filled}}\gets A$
  \State $(A_{\mathrm{filled}})_{ij}\gets v_{0,j}
         \quad\forall(i,j)\in\Omega_2$
  \State \Return $A_{\mathrm{filled}}$
\EndProcedure
\end{algorithmic}
\end{algorithm}
