\begin{algorithm}[ht]
\footnotesize
\captionof{algorithm}{\textbf{Grid-Halving $\beta$--$C$ Search with $K_{\mathrm{CV}}$-Fold Evaluation}}
\label{algo:betaC_Grid_halving}
\begin{algorithmic}[1]
\setlength{\itemsep}{0.3em}

\Require Aligned preprocessed matrices
         $\mathcal D=(X_{\mathrm{ref}},X_{\mathrm{in}})$,
         ordered feature specifications $\Phi$,
         and source missingness mask $\mathbf M$ from
         \cref{algo:data_preprocessing}
\Require Search bounds $\beta_{\min},\beta_{\max},C_{\min},C_{\max}$
\Require Maximum relative grid width $\alpha_0$ (default $0.01$)
\Require GMM component count $K$ and latent dimension $Z$
\Require $K_{\mathrm{CV}}\in\{5,10\}$ folds (default $K_{\mathrm{CV}}=5$),
         $5\leq N\leq20$ imputation draws
         (default $N=10$), batch size $B$, and $5\leq R\leq20$
         imputation-update cycles (default $R=10$), learning rate $\eta$,
         epoch schedule $(E_0,E_{\max},E_{\mathrm{step}})=(30,300,30)$,
         early-stopping tolerance $10^{-4}\leq\tau$ (default $\tau=10^{-3}$),
         and patience $p=5$
\Require Default composite weights $\mathbf w=(5/14,5/14,1/14,1/14,2/14)$
         for multiple-imputation MAE/RMSE and 80\%, 90\%, 95\% coverage
         deviations, respectively
\Ensure Best candidate $q^\star=(\beta^\star,C^\star)$ and its
         fold-averaged score $s^\star$

\State $\mathcal P_{\mathrm{GMM}}\gets
       \textsc{DefineGMMPrior}(K,Z)$
       \Comment{each fold-specific VAE learns its own prior parameters}

\Statex
\Procedure{GridHalvingBetaCSearch}
          {$\mathcal D,\Phi,\mathbf M,\mathcal P_{\mathrm{GMM}},\textsc{config}$}
  \State $(\beta_{\mathrm{lo}}^{(0)},\beta_{\mathrm{hi}}^{(0)})
         \gets(\beta_{\min},\beta_{\max})$
  \State $(C_{\mathrm{lo}}^{(0)},C_{\mathrm{hi}}^{(0)})
         \gets(C_{\min},C_{\max})$
  \State $\Delta\beta_0\gets\beta_{\max}-\beta_{\min};\quad
         \Delta C_0\gets C_{\max}-C_{\min}$
  \State $q^\star\gets\bot;\quad s^\star\gets+\infty;\quad t\gets0$

  \While{$\max\!\left\{
       \dfrac{\beta_{\mathrm{hi}}^{(t)}-\beta_{\mathrm{lo}}^{(t)}}{\Delta\beta_0},
       \dfrac{C_{\mathrm{hi}}^{(t)}-C_{\mathrm{lo}}^{(t)}}{\Delta C_0}
       \right\}>\alpha_0$}
    \State $\beta_{\mathrm{mid}}^{(t)}\gets
           \beta_{\mathrm{lo}}^{(t)}+
           \dfrac{\beta_{\mathrm{hi}}^{(t)}-\beta_{\mathrm{lo}}^{(t)}}{2}$
    \State $C_{\mathrm{mid}}^{(t)}\gets
           C_{\mathrm{lo}}^{(t)}+
           \dfrac{C_{\mathrm{hi}}^{(t)}-C_{\mathrm{lo}}^{(t)}}{2}$
    \State $\mathcal Q_t\gets\{
           (\beta_{\mathrm{lo}}^{(t)},C_{\mathrm{lo}}^{(t)}),
           (\beta_{\mathrm{lo}}^{(t)},C_{\mathrm{hi}}^{(t)}),
           (\beta_{\mathrm{hi}}^{(t)},C_{\mathrm{lo}}^{(t)}),
           (\beta_{\mathrm{hi}}^{(t)},C_{\mathrm{hi}}^{(t)})\}$
    \State $\mathcal R_t\gets
           \textsc{EvaluateCandidatesByKFoldCV}
           (\mathcal Q_t,\mathcal D,\Phi,\mathbf M,
            \mathcal P_{\mathrm{GMM}},N,\textsc{config})$
           \Comment{evaluation procedure in \cref{algo:betaC_candidate_cv}}
    \State $r_t\gets\arg\min_{r\in\mathcal R_t}r.\text{score}$
    \State $(q_t,s_t)\gets(r_t.q,r_t.\text{score})$
    \If{$s_t<s^\star$}
      \State $(q^\star,s^\star)\gets(q_t,s_t)$
    \EndIf

    \State Write $q_t=(\beta_t,C_t)$ and shrink both intervals toward its corner:
    \State $(\beta_{\mathrm{lo}}^{(t+1)},\beta_{\mathrm{hi}}^{(t+1)})\gets
           \begin{cases}
             (\beta_{\mathrm{lo}}^{(t)},\beta_{\mathrm{mid}}^{(t)}),
               & \beta_t=\beta_{\mathrm{lo}}^{(t)},\\
             (\beta_{\mathrm{mid}}^{(t)},\beta_{\mathrm{hi}}^{(t)}),
               & \beta_t=\beta_{\mathrm{hi}}^{(t)}
           \end{cases}$
    \State $(C_{\mathrm{lo}}^{(t+1)},C_{\mathrm{hi}}^{(t+1)})\gets
           \begin{cases}
             (C_{\mathrm{lo}}^{(t)},C_{\mathrm{mid}}^{(t)}),
               & C_t=C_{\mathrm{lo}}^{(t)},\\
             (C_{\mathrm{mid}}^{(t)},C_{\mathrm{hi}}^{(t)}),
               & C_t=C_{\mathrm{hi}}^{(t)}
           \end{cases}$
    \State $t\gets t+1$
  \EndWhile
  \State \Return $(q^\star,s^\star)$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[ht]
\footnotesize
\captionof{algorithm}{\textbf{$K_{\mathrm{CV}}$-Fold Evaluation of a $\beta$--$C$ Candidate}}
\label{algo:betaC_candidate_cv}
\begin{algorithmic}[1]
\setlength{\itemsep}{0.3em}

\Require Candidate set $\mathcal Q$, aligned data $\mathcal D=(X_{\mathrm{ref}},X_{\mathrm{in}})$,
         feature specifications $\Phi$, source mask $\mathbf M$,
         GMM prior specification $\mathcal P_{\mathrm{GMM}}$,
         and $N$ imputation draws
\Ensure Fold-averaged score for each candidate $q\in\mathcal Q$

\Procedure{EvaluateCandidatesByKFoldCV}
          {$\mathcal Q,\mathcal D,\Phi,\mathbf M,
            \mathcal P_{\mathrm{GMM}},N,\textsc{config}$}
  \For{each candidate $q=(\beta,C)\in\mathcal Q$}
    \For{$k=1,\ldots,K_{\mathrm{CV}}$}
      \State Split row indices into disjoint training and validation sets
             $I_k^{\mathrm{tr}}$ and $I_k^{\mathrm{val}}$
      \State $X_{\mathrm{tr}}\gets X_{\mathrm{in}}[I_k^{\mathrm{tr}},:]$
      \State $X_{\mathrm{val}}^{\mathrm{base}}\gets
             X_{\mathrm{in}}[I_k^{\mathrm{val}},:];\quad
             \mathbf M_k\gets\mathbf M[I_k^{\mathrm{val}},:]$
             \Comment{$\mathbf M_k$: 0 observed, 1 source-missing, 2 previously amputed}
      \State $(X_{\mathrm{val}},\mathbf M_k^{\mathrm{val}},
             \mathbf M_k^{\mathrm{imp}})\gets
             \textsc{MakeValidationView}
             (X_{\mathrm{val}}^{\mathrm{base}},\mathbf M_k,\Phi)$
             \Comment{ordinal/categorical groups; other columns singleton; 10\% per unit}
      \State In $\mathbf M_k^{\mathrm{val}}$, selected type-0 cells are labeled 3
             and selected type-1/2 cells are labeled 4; initialize
             $\mathbf M_k^{\mathrm{imp}}$ from $\mathbf M_k$
             \Comment{type 0$\to$3; type 1/2$\to$4; type 4 excluded from scoring}
      \State For each row with no type-3 cell, restore its validation mask from
             $\mathbf M_k$
             \Comment{discard only that row's draw; retain other rows' draws}
      \State Set $X_{\mathrm{val},ij}\gets0$ and
             $\mathbf M_{k,ij}^{\mathrm{imp}}\gets2$
             for cells with $\mathbf M_{k,ij}^{\mathrm{val}}=3$
             \Comment{hide type-3 values; 3$\to$2 for imputation}
      \State Initialize a fresh fold-specific VAE on $X_{\mathrm{tr}}$ using
             $\Phi$, $\mathcal P_{\mathrm{GMM}}$, and $(\beta,C)$
      \State $s_{q,k}^{\star}\gets+\infty;\quad e\gets0;\quad h\gets0$
      \While{$e<E_{\max}$}
        \State Train the VAE for $E_0$ epochs if $e=0$, otherwise for
               $\min(E_{\mathrm{step}},E_{\max}-e)$ epochs; update $e$
        \For{$n=1,\ldots,N$}
          \State $\widehat X_{\mathrm{val}}^{(n)}\gets
                 \textsc{Impute}(X_{\mathrm{val}},
                 \mathbf M_k^{\mathrm{imp}},\Phi;R)$
        \EndFor
        \State $\bar X_{\mathrm{val}}\gets
               \dfrac{1}{N}\sum_{n=1}^{N}\widehat X_{\mathrm{val}}^{(n)}$
        \State Compute type-3 multiple-imputation $\mathrm{MAE}$, $\mathrm{RMSE}$,
               and empirical coverage deviations from
               $\{\widehat X_{\mathrm{val}}^{(n)}\}_{n=1}^N$
               \Comment{exclude type-4 cells}
        \State $s_{q,k}\gets\sum_j w_j v_{q,k,j}$
               \Comment{$\mathbf v_{q,k}$ contains these five type-3 metrics}
        \If{$s_{q,k}<s_{q,k}^{\star}-\tau$}
          \State $s_{q,k}^{\star}\gets s_{q,k};\quad h\gets0$
        \Else
          \State $h\gets h+1$
        \EndIf
        \If{$h\ge p$}
          \State \textbf{break}
        \EndIf
      \EndWhile
      \State \textsc{SaveFoldResult}$(q,k,s_{q,k}^{\star})$
    \EndFor
    \State $\operatorname{score}(q)\gets
           \dfrac{1}{K_{\mathrm{CV}}}
           \sum_{k=1}^{K_{\mathrm{CV}}}s_{q,k}^{\star}$
  \EndFor
  \State \Return candidate scores
\EndProcedure

\end{algorithmic}
\end{algorithm}
